# RULE-BASED AI ENGINE

def calculate_prediction(crop, soil, temperature, rainfall,
                         moisture, fertilizer, irrigation, area):

    data = CROP_DATA[crop]

    score = 0
    advice = []
    risk_factors = []

    # ---------------- Temperature ----------------
    t_min, t_max = data["temp"]

    if t_min <= temperature <= t_max:
        score += 20
    elif abs(temperature - t_min) <= 5 or abs(temperature - t_max) <= 5:
        score += 10
        risk_factors.append("Temperature is slightly outside ideal range.")
        advice.append("Monitor temperature and protect the crop if required.")
    else:
        risk_factors.append("Temperature is not suitable for this crop.")
        advice.append("Try to maintain the recommended temperature conditions.")

    # ---------------- Rainfall ----------------
    r_min, r_max = data["rainfall"]

    if r_min <= rainfall <= r_max:
        score += 20
    elif rainfall < r_min:
        score += 8
        risk_factors.append("Rainfall is below the recommended level.")
        advice.append("Provide additional irrigation if soil becomes dry.")
    elif rainfall > r_max:
        score += 8
        risk_factors.append("Rainfall is higher than the recommended level.")
        advice.append("Ensure proper drainage to avoid waterlogging.")

    # ---------------- Soil Moisture ----------------
    m_min, m_max = data["moisture"]

    if m_min <= moisture <= m_max:
        score += 20
    elif abs(moisture - m_min) <= 10 or abs(moisture - m_max) <= 10:
        score += 10
        risk_factors.append("Soil moisture is near the limit.")
    else:
        score += 5
        risk_factors.append("Soil moisture is unsuitable.")
        advice.append("Adjust irrigation according to soil moisture.")

    # ---------------- Soil Type ----------------
    if soil.lower() == "loamy":
        score += 15
    elif soil.lower() in ["clay", "sandy"]:
        score += 10
    else:
        score += 7

    # ---------------- Fertilizer ----------------
    if fertilizer.lower() == data["fertilizer"].lower():
        score += 10
    elif fertilizer.lower() == "medium":
        score += 7
    else:
        score += 5

    if fertilizer.lower() == "low":
        advice.append("Consider improving fertilizer/nutrient management.")
        risk_factors.append("Fertilizer level is low.")

    # ---------------- Irrigation ----------------
    required_water = data["water"].lower()

    if irrigation.lower() == required_water:
        score += 15
    elif irrigation.lower() == "medium":
        score += 10
    else:
        score += 6

    # Irrigation advice
    if irrigation.lower() == "low":
        advice.append("Increase irrigation frequency if moisture is low.")
    elif irrigation.lower() == "high" and required_water != "high":
        advice.append("Avoid excessive irrigation and waterlogging.")

    # ---------------- Final score ----------------
    score = min(score, 100)

    if score >= 80:
        level = "Excellent"
        condition = "Very Good"
        risk = "Low"
        multiplier = 1.10
    elif score >= 65:
        level = "Good"
        condition = "Good"
        risk = "Low to Medium"
        multiplier = 0.95
    elif score >= 50:
        level = "Average"
        condition = "Moderate"
        risk = "Medium"
        multiplier = 0.80
    else:
        level = "Low"
        condition = "Needs Improvement"
        risk = "High"
        multiplier = 0.60

    estimated_yield = round(data["base_yield"] * multiplier, 2)
    total_production = round(estimated_yield * area, 2)

    # AI Confidence
    confidence = min(95, max(55, score + 5))

    # Smart farming advice
    if not advice:
        advice.append("Current farm conditions are suitable.")
        advice.append("Continue monitoring soil moisture and weather.")
        advice.append("Use balanced fertilizer according to crop requirements.")

    # Soil-specific advice
    if soil.lower() == "sandy":
        advice.append("Sandy soil may require more frequent irrigation.")
    elif soil.lower() == "clay":
        advice.append("Clay soil needs proper drainage.")
    elif soil.lower() == "loamy":
        advice.append("Loamy soil is generally favorable for crop growth.")

    return {
        "crop": data["name"],
        "yield_level": level,
        "estimated_yield": estimated_yield,
        "total_production": total_production,
        "confidence": confidence,
        "score": score,
        "condition": condition,
        "risk": risk,
        "risk_factors": risk_factors,
        "advice": advice
    }

function getFormData() {

    return {
        crop: document.getElementById("crop").value,

}


async function predict() {

    const data = getFormData();

    if (
        !data.temperature ||
        !data.rainfall ||
        !data.moisture ||
        !data.area
    ) {
        alert("Please enter all required values.");
        return;
    }

    try {

        const response = await fetch("/predict", {

            method: "POST",

            headers: {
                "Content-Type": "application/json"
            },

            body: JSON.stringify(data)

        });

        const result = await response.json();

        latestResult = result;

        displayResult(result);

    } catch(error) {

        alert("Error connecting to AI server.");

        console.log(error);

    }

}

// VOICE INPUT

function startVoice() {

    const SpeechRecognition =
        window.SpeechRecognition ||
        window.webkitSpeechRecognition;

    if (!SpeechRecognition) {

        alert(
            "Voice recognition is not supported in this browser. Please use Google Chrome."
        );

        return;
    }

    const recognition = new SpeechRecognition();

    recognition.lang = "en-IN";

    recognition.continuous = false;

    recognition.interimResults = false;


    document.getElementById("voiceStatus").innerText =
        "🎤 Listening... Please speak your farm information.";


    recognition.start();


    recognition.onresult = function(event) {

        const text =
            event.results[0][0].transcript.toLowerCase();

        document.getElementById("voiceStatus").innerText =
            "You said: " + text;

        processVoiceCommand(text);

    };


    recognition.onerror = function(event) {

        document.getElementById("voiceStatus").innerText =
            "Voice error: " + event.error;

    };


    recognition.onend = function() {

        console.log("Voice recognition ended.");

    };

}

// VOICE COMMAND PARSER

function processVoiceCommand(text) {

    // Crop
    if (text.includes("rice")) {
        document.getElementById("crop").value = "rice";
    }
    else if (text.includes("wheat")) {
        document.getElementById("crop").value = "wheat";
    }
    else if (text.includes("maize") || text.includes("corn")) {
        document.getElementById("crop").value = "maize";
    }
    else if (text.includes("potato")) {
        document.getElementById("crop").value = "potato";
    }
    else if (text.includes("sugarcane")) {
        document.getElementById("crop").value = "sugarcane";
    }


    // Soil
    if (text.includes("loamy")) {
        document.getElementById("soil").value = "loamy";
    }
    else if (text.includes("clay")) {
        document.getElementById("soil").value = "clay";
    }
    else if (text.includes("sandy")) {
        document.getElementById("soil").value = "sandy";
    }


    // Fertilizer
    if (text.includes("high fertilizer")) {
        document.getElementById("fertilizer").value = "high";
    }
    else if (text.includes("low fertilizer")) {
        document.getElementById("fertilizer").value = "low";
    }
    else if (text.includes("medium fertilizer")) {
        document.getElementById("fertilizer").value = "medium";
    }


    // Irrigation
    if (text.includes("high irrigation")) {
        document.getElementById("irrigation").value = "high";
    }
    else if (text.includes("low irrigation")) {
        document.getElementById("irrigation").value = "low";
    }
    else if (text.includes("medium irrigation")) {
        document.getElementById("irrigation").value = "medium";
    }


    // Numbers
    const numbers =
        text.match(/-?\\d+(?:\\.\\d+)?/g);


    if (numbers && numbers.length >= 1) {

        document.getElementById("temperature").value =
            numbers[0];

    }

    if (numbers && numbers.length >= 2) {

        document.getElementById("rainfall").value =
            numbers[1];

    }

    if (numbers && numbers.length >= 3) {

        document.getElementById("moisture").value =
            numbers[2];

    }

    if (numbers && numbers.length >= 4) {

        document.getElementById("area").value =
            numbers[3];

    }


    setTimeout(function() {

        predict();

    }, 700);

}
